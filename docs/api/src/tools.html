 62def load_tables() -> dict[str, pd.DataFrame]:
 63    
 64    categories = load_csv("categories")
 65    customers = load_csv("customers")
 66    orders = load_csv("orders")
 67    order_items = load_csv("order_items")
 68    products = load_csv("products")
 69    promotions = load_csv("promotions")
 70
 71    # Login compares these, so people can type the phone with or without (67), spaces, dashes.
 72    customers["email"] = customers["email"].str.strip().str.lower()
 73    customers["phone_digits"] = customers["phone"].str.replace(r"\D", "", regex=True)
 74
 75    # Orders that haven't shipped yet have no estimated_delivery, those stay empty (NaT).
 76    for col in ("order_date", "estimated_delivery"):
 77        orders[col] = pd.to_datetime(orders[col])

 79    # specs comes as a JSON string in the CSV.
 80    products["specs"] = products["specs"].map(json.loads)
 81    products = products.merge(

 83    )
 84    # Accent-free copy of name + description + category, just for searching.
 85    products["search_text"] = (products["name"] + " " + products["description"] + " " + products["category"]).map(normalize)
 86
 87    promotions["is_active"] = promotions["is_active"].astype(bool)
 88
 89    return {
 90        "categories": categories,
 91        "customers": customers,
 92        "orders": orders,

100def load_policy() -> dict[str, str]:
101    
102    text = " ".join(page.extract_text() for page in PdfReader(POLICY_FILE).pages)
103    # pypdf returns this file with almost one word per line, so I flatten the whitespace first.
104    text = re.sub(r"\s+", " ", text)
105    text = text.replace("Empório da Música Manual de Políticas e Procedimentos", "")  # page header
106    text = re.sub(r"Página \d+", "", text)  # page footer
107    text = re.sub(r"(\d) - (\d)", r"\1-\2", text)  # phone numbers split in the tables, "3341 - 4444"
108    text = re.sub(r"\s+", " ", text).strip()
109
110    starts = [text.index(heading) for heading in POLICY_HEADINGS.values()] + [len(text)]
111    sections = {key: text[starts[i]:starts[i + 1]].strip() for i, key in enumerate(POLICY_HEADINGS)}
112    # The cover page (address, phone, e-mail) comes before section 1, so I keep it with "about".
113    sections["about"] = text[: starts[0]].strip() + "\n" + sections["about"]
114    return sections

147    promos = tables()["promotions"]

155#  Most interest-free installments allowed for this price (manual 3.1). Minimum installment: R$ 50 up to 3x, R$ 80 from 4x to 6x, R$ 100 from 7x to 12x.   
156def max_installments(price: float) -> int:
157   
158    for n in range(12, 1, -1):
159        minimum = 100 if n >= 7 else 80 if n >= 4 else 50
160        if price / n >= minimum:
161            return n
162    return 1

165def price_info(product_id: int, list_price: float) -> dict:
166    """Every price figure the agent is allowed to quote for a product."""
167    promo = active_discount(product_id)
168    if promo:
169        percent, campaign = promo
170        final = round(list_price * (1 - percent / 100), 2)
171        info = {"list_price_brl": list_price, "promotion": campaign, "discount_percent": percent,
172                "final_price_brl": final, "pix_price_brl": None}  # PIX doesn't apply on top of a promo
173    else:
174        final = list_price
175        info = {"list_price_brl": list_price, "final_price_brl": final,
176                "pix_price_brl": round(list_price * (1 - PIX_DISCOUNT), 2)}
177        # Every installment option the customer can pick, so "em 6x?" or "em 9x?" is a lookup, never a division.
178    options = {f"{n}x": round(final / n, 2) for n in range(1, max_installments(final) + 1)}
179    return {**info, "installments": options}

191def search_products(category: str | None = None, max_price: float | None = None, term: str | None = None,
192                    sort: str = "cheapest") -> dict:
193    """Products filtered by category, max final price and words in name/description. All optional.
194
195    sort is "cheapest" or "most_expensive".
196    """
197    products = tables()["products"]
198    if category and normalize(category) not in ANY_CATEGORY:
199        products = products[products["category"].map(normalize) == normalize(category)]
200    if term:
201        for word in normalize(term).split():  # every word has to show up somewhere
202            if word in FILLER_WORDS:
203                continue
204            products = products[products["search_text"].str.contains(word, regex=False)]
205
206    results = []
207    for p in products.itertuples():
208        info = price_info(int(p.product_id), float(p.price_brl))
209        item = {
210            "product_id": int(p.product_id),
211            "name": p.name,
212            "category": p.category,
213            "availability": availability(p.status, p.stock_quantity),
214            "final_price_brl": info["final_price_brl"],
215        }
216        if "promotion" in info:
217            item["discount_percent"] = info["discount_percent"]
218        # max_price is checked on the final price, so a product on sale can fit under the limit
219        if max_price is None or item["final_price_brl"] <= max_price:
220            results.append(item)
221
222
223    # In-stock first, then by price. Otherwise the 10-item cut could hide products people can buy.
224    sign = -1 if sort == "most_expensive" else 1
225    results.sort(key=lambda r: (r["availability"] != "in_stock", sign * r["final_price_brl"]))
226    found = {"total_found": len(results), "products": results[:MAX_RESULTS]}
227    # Nothing found can be bought: hand over real alternatives, or a small model makes some up.
228    if results and all(r["availability"] != "in_stock" for r in results):
229        found["in_stock_alternatives"] = similar_in_stock(results[0])
230    return found

232def similar_in_stock(product: dict, limit: int = 3) -> list[dict]:
233    """In-stock products from the same category, closest in price to the given one."""
234    products = tables()["products"]
235    in_stock = products[(products["category"] == product["category"])
236                        & (products["status"] == "active") & (products["stock_quantity"] > 0)]
237    candidates = [
238        {"product_id": int(p.product_id), "name": p.name,
239         "final_price_brl": price_info(int(p.product_id), float(p.price_brl))["final_price_brl"]}
240        for p in in_stock.itertuples()
241    ]
242    candidates.sort(key=lambda c: abs(c["final_price_brl"] - product["final_price_brl"]))
243    return candidates[:limit]

245def get_product(product_id: int) -> dict:
246    """Everything about one product: prices, every installment option, stock and specs."""
247    products = tables()["products"]
248    row = products[products["product_id"] == int(product_id)]
249    if row.empty:
250        return {"error": f"No product with id {product_id}."}
251    p = row.iloc[0]
252    return {
253        "product_id": int(p["product_id"]),
254        "name": p["name"],
255        "category": p["category"],
256        "availability": availability(p["status"], p["stock_quantity"]),
257        "stock_quantity": int(p["stock_quantity"]),
258        **price_info(int(p["product_id"]), float(p["price_brl"])),
259        "description": p["description"],
260        "specs": p["specs"],
261    }

310def _matches(customers: pd.DataFrame, email_or_phone: str) -> pd.Series:
311    """Rows whose e-mail is exactly this, or whose phone ends with these digits (8 or more)."""
312    given = email_or_phone.strip().lower()
313    digits = "".join(ch for ch in given if ch.isdigit())
314    by_email = customers["email"] == given
315    by_phone = customers["phone_digits"].str.endswith(digits) if len(digits) >= 8 else False
316    return by_email | by_phone

319def find_customer(email_or_phone: str, confirm: str | None = None) -> dict:
320    """Login lookup, used by /login. Never exposed to the model.
321
322    Returns {"status": "ok", "customer_id", "name"} when exactly one record matches.
323    When more than one does (customers 11 and 43 share an e-mail), it returns
324    {"status": "need_more", "ask_for": "phone" or "email"} so the CLI can ask for the other contact,
325    and the second call passes it in confirm. Anything else is {"status": "not_found"}.
326    """
327    customers = tables()["customers"]
328    found = customers[_matches(customers, email_or_phone)]
329    if confirm is not None:
330        found = found[_matches(found, confirm)]  # the second contact must belong to one of the first hits
331    if len(found) == 1:
332        c = found.iloc[0]
333        return {"status": "ok", "customer_id": int(c["customer_id"]), "name": c["name"]}
334    if len(found) > 1 and confirm is None:
335        return {"status": "need_more", "ask_for": "phone" if "@" in email_or_phone else "email"}
336    return {"status": "not_found"}

338def get_order(order_id: int, customer_id: int | None) -> dict:
339    """One order of the logged-in customer: status, tracking, dates and items."""
340    if customer_id is None:
341        return NOT_LOGGED_IN
342    t = tables()
343    orders = t["orders"]
344    # The owner check is part of the filter: someone else's order just isn't found.
345    match = orders[(orders["order_id"] == int(order_id)) & (orders["customer_id"] == customer_id)]
346    if match.empty:
347        return ORDER_NOT_FOUND
348    order = match.iloc[0]
349
350    items = t["order_items"].merge(t["products"], on="product_id")
351    items = items[items["order_id"] == order["order_id"]]
352
353    # Date math happens here. Small models are bad at it and called an 8-month-late delivery "recent".
354    today = date.today()
355    ordered = order["order_date"].date()
356    eta = None if pd.isna(order["estimated_delivery"]) else order["estimated_delivery"].date()
357    return {
358        "order_id": int(order["order_id"]),
359        "status": order["status"],  # pending, confirmed, shipped, delivered or cancelled
360        "order_date": ordered.isoformat(),
361        "days_since_order": (today - ordered).days,
362        "estimated_delivery": eta.isoformat() if eta else None,
363        "days_since_estimated_delivery": (today - eta).days if eta else None,
364        "delivery_overdue": bool(order["status"] == "shipped" and eta is not None and eta < today),
365        "tracking_code": None if pd.isna(order["tracking_code"]) else order["tracking_code"],
366        "payment_method": order["payment_method"],
367        "total_brl": float(order["total_brl"]),
368        "items": [{"name": i.name, "quantity": int(i.quantity)} for i in items.itertuples()],
369        "notes": None if pd.isna(order["notes"]) else order["notes"],
370    }

def price_info(product_id: int, list_price: float) -> dict: 
166def price_info(product_id: int, list_price: float) -> dict:
167    """Every price figure the agent is allowed to quote for a product."""
168    promo = active_discount(product_id)
169    if promo:
170        percent, campaign = promo
171        final = round(list_price * (1 - percent / 100), 2)
172        info = {"list_price_brl": list_price, "promotion": campaign, "discount_percent": percent,
173                "final_price_brl": final, "pix_price_brl": None}  # PIX doesn't apply on top of a promo
174    else:
175        final = list_price
176        info = {"list_price_brl": list_price, "final_price_brl": final,
177                "pix_price_brl": round(list_price * (1 - PIX_DISCOUNT), 2)}
178        # Every installment option the customer can pick, so "em 6x?" or "em 9x?" is a lookup, never a division.
179    options = {f"{n}x": round(final / n, 2) for n in range(1, max_installments(final) + 1)}
180    return {**info, "installments": options}

def search_products( category: str | None = None, max_price: float | None = None, term: str | None = None, sort: str = 'cheapest') -> dict: 
192def search_products(category: str | None = None, max_price: float | None = None, term: str | None = None,
193                    sort: str = "cheapest") -> dict:
194    """Products filtered by category, max final price and words in name/description. All optional.
195
196    sort is "cheapest" or "most_expensive".
197    """
198    products = tables()["products"]
199    if category and normalize(category) not in ANY_CATEGORY:
200        products = products[products["category"].map(normalize) == normalize(category)]
201    if term:
202        for word in normalize(term).split():  # every word has to show up somewhere
203            if word in FILLER_WORDS:
204                continue
205            products = products[products["search_text"].str.contains(word, regex=False)]
206
207    results = []
208    for p in products.itertuples():
209        info = price_info(int(p.product_id), float(p.price_brl))
210        item = {
211            "product_id": int(p.product_id),
212            "name": p.name,
213            "category": p.category,
214            "availability": availability(p.status, p.stock_quantity),
215            "final_price_brl": info["final_price_brl"],
216        }
217        if "promotion" in info:
218            item["discount_percent"] = info["discount_percent"]
219        # max_price is checked on the final price, so a product on sale can fit under the limit
220        if max_price is None or item["final_price_brl"] <= max_price:
221            results.append(item)
222
223
224    # In-stock first, then by price. Otherwise the 10-item cut could hide products people can buy.
225    sign = -1 if sort == "most_expensive" else 1
226    results.sort(key=lambda r: (r["availability"] != "in_stock", sign * r["final_price_brl"]))
227    found = {"total_found": len(results), "products": results[:MAX_RESULTS]}
228    # Nothing found can be bought: hand over real alternatives, or a small model makes some up.
229    if results and all(r["availability"] != "in_stock" for r in results):
230        found["in_stock_alternatives"] = similar_in_stock(results[0])
231    return found

233def similar_in_stock(product: dict, limit: int = 3) -> list[dict]:
234    """In-stock products from the same category, closest in price to the given one."""
235    products = tables()["products"]
236    in_stock = products[(products["category"] == product["category"])
237                        & (products["status"] == "active") & (products["stock_quantity"] > 0)]
238    candidates = [
239        {"product_id": int(p.product_id), "name": p.name,
240         "final_price_brl": price_info(int(p.product_id), float(p.price_brl))["final_price_brl"]}
241        for p in in_stock.itertuples()
242    ]
243    candidates.sort(key=lambda c: abs(c["final_price_brl"] - product["final_price_brl"]))
244    return candidates[:limit]

def find_customer(email_or_phone: str, confirm: str | None = None) -> dict: 
320def find_customer(email_or_phone: str, confirm: str | None = None) -> dict:
321    """Login lookup, used by /login. Never exposed to the model.
322
323    Returns {"status": "ok", "customer_id", "name"} when exactly one record matches.
324    When more than one does (customers 11 and 43 share an e-mail), it returns
325    {"status": "need_more", "ask_for": "phone" or "email"} so the CLI can ask for the other contact,
326    and the second call passes it in confirm. Anything else is {"status": "not_found"}.
327    """
328    customers = tables()["customers"]
329    found = customers[_matches(customers, email_or_phone)]
330    if confirm is not None:
331        found = found[_matches(found, confirm)]  # the second contact must belong to one of the first hits
332    if len(found) == 1:
333        c = found.iloc[0]
334        return {"status": "ok", "customer_id": int(c["customer_id"]), "name": c["name"]}
335    if len(found) > 1 and confirm is None:
336        return {"status": "need_more", "ask_for": "phone" if "@" in email_or_phone else "email"}
337    return {"status": "not_found"}

339def get_order(order_id: int, customer_id: int | None) -> dict:
340    """One order of the logged-in customer: status, tracking, dates and items."""
341    if customer_id is None:
342        return NOT_LOGGED_IN

344    orders = t["orders"]
345    # The owner check is part of the filter: someone else's order just isn't found.

350

363        "estimated_delivery": eta.isoformat() if eta else None,
